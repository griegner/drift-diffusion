\section{Methods}

\subsection{Notation}

\textbf{Drift Diffusion Model}
\begin{align}
    Z_\tau = az + \sum_{t=1}^\tau e_t, \quad e_t \sim \mathcal{N}(v\Delta\tau, \sigma^2\Delta\tau)
\end{align}
$a > 0$ is the decision boundary\\
$z \in []$

$v \in \mathbb{R}$ is the drift rate\\
discrete-time random walk, $\tau$ indexes time on a temporal grid with resolution $\Delta \tau$, $\Delta \tau \to 0$ is a continuous \textit{drift diffusion} process (Wiener diffusion process if the errors are Gaussian)\\
$Z_t$ is a decision variable that captures stochastic evidence accumulation in favor of a given option in a binary choice set\\
Gaussian diffusion\\


\textbf{RT + Response for} $\{Z_\tau: \tau=0, .., RT\}$
\begin{align}
    RT = \begin{cases}
        + \text{min}\{\tau>0: Z_\tau \ge +a\}\\
        - \text{min}\{\tau>0: Z_\tau \le -a\}
        \end{cases}
\end{align}
$|RT|> 0$ is the reaction time\\
$\text{sign}(RT) \in \{-1, +1\}$ is the binary response (change to choice?)\\
a decision is made at the first time $\tau$ for which $Z_\tau$ hits either decision boundary\\
(add non-decision time $t_0$, the portion of the reaction time that occurs independently of decision-making - after some physiological delay $t_0$ the action is executed)\\
single process explains both which choice is made ($\text{sign}(RT)$) and how long it takes to make it ($|RT|$)\\
the four DDM parameters, namely the drift rate $v$, decision boundary $a$, relative starting point $z$, and non-decision time $t_0$ will be grouped under a single vector $\theta = (v, a, z, t_0)$

\textbf{RTs + Responses for} $\{Z_\tau: \tau=0, .., RT\}_t^T$
\begin{align}
    RT_t = \{RT_1, ..., RT_T\} \sim \mathcal{D}(a, v)
\end{align}

$\tau \in 1:RT$ is the diffusion time index, $t \in 1:T$ is the trial time index\\
$t > 0$ is the trial time index\\
$\Delta t$ is nonconstant (unequal sampling)\\
$\mathcal{D}$ is probability distribution determined by parameters $a$ and $v$ (choice-conditional distribution of RTs)\\
joint reaction time/response distributions\\
certain parameter settings will produce faster but more entropic response distributions, whereas others will produce slower, more direct responses

\subsection{Definitions and Estimation}
\textbf{Probability Density Function}

\begin{align}
    f(RT \mid a, v) = \frac{\pi}{a^2}\exp\left(-\frac{va}{2}-\frac{v^2t}{2}\right) \times\sum_{k=1}^\infty k\exp\left(-\frac{k^2\pi^2 RT}{2a^2}\right)\sin(\frac{k\pi}{2})
\end{align}

probability of $Z_\tau$ crossing either of the decision boundaries ($\pm a$) at any specified time $RT$.\\
the probability of the reaction time and choice pair $\{|RT|, \text{sign}(RT)\}$ is the probability of the stochastic diffusion processing reaching a boundary $\pm a$ at time $|RT|$\\
conditional probability induced by equations (1, 2)\\
unimodal and right-skewed\\
analytic expression derived by Feller (1968m ch. 14, Eq, 6.15) and a finite approximation of the infinite series expansion provided by Navarro\\
involves an infinite sum, and must therefore be implemented through approximations

\textbf{Likelihood Function}
\begin{align}
    L_T(a, v \mid RT_t) = f(RT_t, ..., RT_T \mid a, v) = \prod_{t=1}^T f(RT_t \mid a, v)
\end{align}

\textbf{Log-Likelihood Function}
\begin{align}
    \ell_T (a, v \mid RT_t) = \log L_T(a, v \mid RT_t) = \sum_{t=1}^T \log f(RT_t \mid a, v)
\end{align}

\textbf{MLE} $\hat\theta$ of $\theta = (a, v)$
\begin{align}
    \hat\theta = \underset{\theta}{\text{argmin}} -\ell_T (\theta | RT_t)
\end{align}

\subsection{Asymptotic Covariance Estimators}

\begin{align}
    \widehat{\mathcal{H}}_\theta &= \frac{1}{T}\sum_{t=1}^T-\frac{\partial^2}{\partial\theta\partial\theta^{\prime}}\log f(RT_t\mid\hat{\theta})=-\frac{1}{T}\frac{\partial^2}{\partial\theta\partial\theta^{\prime}}\ell_T(\hat{\theta})\\
    \widehat{\mathscr{I}}_\theta &= \frac{1}{T}\sum_{t=1}^T\left(\frac{\partial}{\partial\theta}\log f(RT_t\mid\hat{\theta})\right)\left(\frac{\partial}{\partial\theta}\log f(RT_t\mid\hat{\theta})\right)^{\prime} = \frac{1}{T} \sum_{t=1}^T \hat{S_t} \hat{S_t}^\prime
\end{align}

\textbf{(1) Sample Hessian}\\
\begin{align}
    \widehat V_1 = \widehat{\mathcal{H}}_\theta^{-1}
\end{align}
\textbf{(2) Outer Product}\\
\begin{align}
    \widehat V_2 = \widehat{\mathscr{I}}_\theta^{-1}
\end{align}
\textbf{(3) Misspecification Robust} $\mathscr{I}_\theta \ne \mathcal{H}_\theta$\\
\begin{align}
    \widehat V_3 = \widehat{\mathcal{H}}_\theta^{-1} \widehat{\mathscr{I}}_\theta \widehat{\mathcal{H}}_\theta^{-1}
\end{align}
\textbf{(4) Autocorrelation Robust}\\
\begin{align}
    \widehat V_4 = \widehat{\mathcal{H}}_\theta^{-1} \left(\frac{1}{T} \sum_{i,j=1}^T w_{|i-j|} \hat{S_i} \hat{S_j}^\prime\right) \widehat{\mathcal{H}}_\theta^{-1}
\end{align}
